# Daniel's DualSense default - loaded automatically at every JSM startup.
# Ported from the Windows config; see the notes marked CHANGED / NOTE.

# ---- Required ----
RESET_MAPPINGS
TELEMETRY_ENABLED = ON
TELEMETRY_PORT = 8974

# ---- Gyro behaviour ----
# TICK_TIME = 1 now actually polls at 1ms on macOS: the port implements
# pacing with mach_wait_until instead of a plain sleep, which the system
# would otherwise round up to ~15ms.
TICK_TIME = 1
REAL_WORLD_CALIBRATION = 151.1
IN_GAME_SENS = 10
IGNORE_GYRO_DEVICES = 0x045e:0x028e

# ---- Noise & steadying ----
GYRO_CUTOFF_SPEED = 0.4
GYRO_SMOOTHING_DECAY = ON
GYRO_CUTOFF_RECOVERY = 0.5

# ---- Sensitivity ----
ROLL_CONTRIBUTION = -100
MIN_GYRO_SENS = 5 3
MAX_GYRO_SENS = 21 6
MIN_GYRO_THRESHOLD = 0
ACCEL_CURVE = QUADRATIC
MAX_GYRO_THRESHOLD = 80

# ---- Face buttons ----
S = SPACE
E = V
N = Q
W = R

# ---- D-pad ----
RIGHT = SCROLLUP
DOWN = C
UP = TAB
LEFT = N

# ---- Bumpers ----
L = LSHIFT
R = E

# ---- Triggers ----
# NOTE: adaptive trigger effects are only exposed over USB. Over Bluetooth
# the DualSense does not advertise trigger rumble, so this is inert there.
ADAPTIVE_TRIGGER = ON
TRIGGER_THRESHOLD = 0
ZL = RMOUSE
ZR = LMOUSE

# ---- Centre buttons ----
# The command names really are "-" and "+", not MINUS/PLUS: ButtonID streams
# them as those characters. "-" is Create/Share, "+" is Options. Your paste
# showed these as "*" because markdown ate the leading - and +.
- = ESC
+ = H

MIC = F

# ---- Gyro ratcheting ----
# CHANGED: was GYRO_OFF = RSR, a DualSense *Edge* grip paddle that your
# standard DualSense (product 0x0ce6) does not have, so gyro could never be
# paused. Now on R1: hold to suspend gyro and reposition your hands.
# R1 keeps its E binding too - the gyro gate reads the button directly.
GYRO_OFF = R

# CHANGED: dropped "LSL = LSHIFT" - also an Edge-only paddle, and L (L1)
# already sends LSHIFT, so nothing is lost.

# ---- Stick bindings ----
LUP = W
LDOWN = S
LLEFT = A
LRIGHT = D
L3 = F
RUP = X
RDOWN = Z
RLEFT = MMOUSE
R3 = LCONTROL

# ---- Sticks ----
LEFT_RING_MODE = INNER
RIGHT_STICK_DEADZONE_INNER = 0.9

# ---- Misc ----
# NOTE: no-op on macOS (getMouseSpeed() always returns 1.0 and nothing in the
# CGEvent path applies an acceleration curve). Kept because it means your
# Windows REAL_WORLD_CALIBRATION transfers over exactly.
IGNORE_OS_MOUSE_SPEED

# NOTE: Windows-only console behaviour; harmless here.
HIDE_MINIMIZED = ON

AUTO_CALIBRATE_GYRO = OFF
LEFT_TRIGGER_OFFSET = 60
LEFT_TRIGGER_RANGE = 255
RIGHT_TRIGGER_OFFSET = 60
RIGHT_TRIGGER_RANGE = 255

# CHANGED: dropped "GYRO_SMOOTH_DECAY = ON" - not a real setting, and a
# duplicate of GYRO_SMOOTHING_DECAY which is already ON above.
# CHANGED: dropped "JOYCON_GYRO_SOURCE = RIGHT" - not a real setting. The
# real one is JOYCON_GYRO_MASK = IGNORE_LEFT, and it only affects Joy-Cons,
# so it does nothing for a DualSense.
